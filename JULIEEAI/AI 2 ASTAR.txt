import heapq

goal = [[1,2,3],
        [4,5,6],
        [7,8,0]]

def print_puzzle(state):
    print()
    for row in state:
        print("|", end=" ")
        for val in row:
            print(val if val != 0 else "_", end=" | ")
        print()
    print()


def heuristic(state):
    # misplaced tiles
    count = 0
    for i in range(3):
        for j in range(3):
            if state[i][j] != 0 and state[i][j] != goal[i][j]:
                count += 1
    return count


def find_blank(state):
    for i in range(3):
        for j in range(3):
            if state[i][j] == 0:
                return i, j


def state_to_tuple(state):
    return tuple(tuple(row) for row in state)


def a_star(start):
    pq = []
    visited = set()
    counter = 0   # tie-breaker

    heapq.heappush(pq, (0, counter, start, []))

    while pq:
        cost, _, state, path = heapq.heappop(pq)

        state_key = state_to_tuple(state)
        if state_key in visited:
            continue

        visited.add(state_key)
        path = path + [state]

        if state == goal:
            print("\nGOAL REACHED!\n")
            print("Solution Path:\n")
            for step in path:
                print_puzzle(step)
            return

        x, y = find_blank(state)

        moves = [(0,1),(0,-1),(1,0),(-1,0)]

        for dx, dy in moves:
            nx, ny = x+dx, y+dy

            if 0 <= nx < 3 and 0 <= ny < 3:
                new_state = [row[:] for row in state]
                new_state[x][y], new_state[nx][ny] = new_state[nx][ny], new_state[x][y]

                new_key = state_to_tuple(new_state)

                if new_key not in visited:
                    g = len(path)
                    h = heuristic(new_state)
                    counter += 1
                    heapq.heappush(pq, (g+h, counter, new_state, path))

📘 THEORY
The A* (A-Star) Algorithm is a heuristic search algorithm used in Artificial Intelligence for finding the shortest and optimal path between initial and goal states. It combines the advantages of Uniform Cost Search and Greedy Best First Search.
The algorithm uses the evaluation function:
f(n)=g(n)+h(n)f(n) = g(n) + h(n)f(n)=g(n)+h(n)
Where:


g(n) → Actual cost from start node to current node


h(n) → Heuristic estimated cost from current node to goal


f(n) → Total estimated cost


In this practical, the 8-puzzle problem is solved using A* algorithm with the heuristic function “Misplaced Tiles.” The algorithm explores puzzle states and selects the state with minimum cost until the goal state is reached.
Priority Queue (heapq) is used to efficiently select the state with lowest cost.
