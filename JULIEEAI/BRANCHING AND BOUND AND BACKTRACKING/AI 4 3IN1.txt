# N-Queens Problem (Backtracking)
def solve_nqueens(n):
    solutions = []
    def place(row, cols, diag1, diag2, board):
        if row == n:
            solutions.append(board[:])
            return
        for col in range(n):
            if col in cols or (row - col) in diag1 or (row + col) in diag2:
                continue
            board.append(col)
            place(row + 1, cols | {col},
                  diag1 | {row - col}, diag2 | {row + col}, board)
            board.pop()                       # backtrack
    place(0, set(), set(), set(), [])
    return solutions
 
def print_board(solution, n):
    for col in solution:
        print(". " * col + "Q " + ". " * (n - col - 1))
    print()
 
# Graph Coloring Problem (Backtracking)
def graph_coloring(graph, m):
    nodes = list(graph)
    color = {}
    def assign(i):
        if i == len(nodes):
            return True
        node = nodes[i]
        for c in range(1, m + 1):
            if all(color.get(nb) != c for nb in graph[node]):
                color[node] = c
                if assign(i + 1):
                    return True
                del color[node]               # backtrack
        return False
    return color if assign(0) else None
 
if __name__ == "__main__":
    n = 4
    solutions = solve_nqueens(n)
    print(f"{n}-Queens: {len(solutions)} solutions found.")
    print("First solution:")
    print_board(solutions[0], n)
 
    graph = {0:[1,2,3], 1:[0,2], 2:[0,1,3], 3:[0,2]}
    result = graph_coloring(graph, 3)
    print("Graph Coloring with 3 colors:", result)

📘 THEORY

Constraint Satisfaction Problems (CSP) are problems where variables must be assigned values while satisfying given constraints. Backtracking is a recursive search technique used to solve CSPs by trying possible solutions and undoing invalid choices.

The N-Queens problem involves placing N queens on an N×N chessboard such that no two queens attack each other. Constraints include:

No same row
No same column
No same diagonal

The Graph Coloring problem assigns colors to graph vertices such that no two adjacent vertices have the same color.

In this practical, Python recursion and backtracking were used to efficiently explore valid configurations and reject invalid states.

✅ CONCLUSION

The N-Queens and Graph Coloring problems were successfully solved using Backtracking technique in Python. The algorithm explored possible configurations recursively and backtracked whenever constraints were violated.
