# Sorting of Numbers Program

numbers = []

n = int(input("How many numbers do you want to enter? "))

for i in range(n):

    num = int(input("Enter number: "))

    numbers.append(num)

print("\nBefore Sorting:")
print(numbers)

# Sorting
numbers.sort()

print("\nAfter Sorting:")
print(numbers)
