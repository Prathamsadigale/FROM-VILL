code : apex soting of numbers  

public class SortNumbersExample {

    // Method to sort numbers
    public static void sortNumbers() {

        // Create list of integers
        List<Integer> numbers = new List<Integer>{5, 2, 9, 1, 7};

        // Print original list
        System.debug('Before Sorting: ' + numbers);

        // Sort the numbers in ascending order
        numbers.sort();

        // Print sorted list
        System.debug('After Sorting: ' + numbers);
    }
}
